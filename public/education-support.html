document.getElementById('eduForm').addEventListener('submit', async (e) => { e.preventDefault(); const btn = document.getElementById('submitBtn'); btn.disabled = true; btn.textContent = 'Processing...'; const examSelect = document.getElementById('examSelect'); const examType = activeAction === 'BUY' ? examSelect.value : 'DONATION_FUND'; const serviceTitle = activeAction === 'BUY' ? examSelect.options[examSelect.selectedIndex].text.split(' — ')[0] : 'Education Sponsorship Fund'; const targetInput = document.getElementById('targetInput').value; const amount = parseFloat(document.getElementById('amount').value); try { if (selectedPayMethod === 'WALLET') { // 1. Gateway Wallet Deduction (Instant Settlement) const merchant = JSON.parse(localStorage.getItem('merchantSession') || '{}'); const res = await fetch('/api/v1/checkout/wallet', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({ merchantPhone: merchant.phone, serviceType: examType, targetInput: targetInput, amount: amount }) }); const data = await res.json(); if (data.status === 'success') { triggerPrintableReceipt({ orderRef: data.orderRef, serviceTitle: serviceTitle, beneficiary: targetInput, cleanAmount: amount - 20, feeAmount: 20.00, totalCharged: amount, tokenOrPin: data.pinToken || 'N/A', serialNo: data.pinSerial || 'N/A', paymentMethod: 'Gateway Wallet Balance' }); } else { alert(data.message || 'Insufficient wallet balance.'); } } else { // 2. Bank Transfer or Card Payment (Pending Payment Confirmation) const res = await fetch('/api/v1/checkout/initialize', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({ serviceType: examType, targetInput: targetInput, amount: amount - 20, paymentMethod: selectedPayMethod }) }); const data = await res.json(); if (data.status === 'success') { if (data.paymentMethod === 'TRANSFER') { // Display Account Transfer Instructions (DO NOT SHOW SUCCESS RECEIPT YET) document.getElementById('modalBody').innerHTML = `

🏦 Complete Bank Transfer

Please transfer the exact amount below to complete your order:

Bank Name: ${data.bankDetails.bankName}

${data.bankDetails.accountNumber}

Amount to Pay: ${data.bankDetails.amountToPay}

Order Ref: ${data.orderRef}

⚡ e-PIN / Service will be dispatched automatically as soon as bank payment is confirmed!
`; document.getElementById('customModal').style.display = 'flex'; } else { window.location.href = data.checkoutUrl; } } } } catch (err) { alert("❌ Network connection error. Please try again."); } btn.disabled = false; btn.textContent = activeAction === 'BUY' ? 'Proceed to Buy e-PIN' : 'Donate to Education Support Fund'; });
